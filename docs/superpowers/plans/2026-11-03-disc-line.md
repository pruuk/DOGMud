> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild quests 75, 76 and 77 as a hard, fair mystery that walks east to a dead man's camp, add scattered torn novel pages and four achievements, raise the crash site's entry to 400 gold, and give players ways to find the start.

**Architecture:** Two small engine additions (a saved set of items read, two new achievement trigger types). Everything else is content on existing systems: quest triggers, dialogue topics, shop lists, `knows_facts` gossip. Four PRs in order, each working alone.

**Tech Stack:** Go, YAML world data, `questengine` walker tests, `tools/id_inventory.py`, `cmd/dialoguecheck`, the playtest harness.

**Spec:** `docs/superpowers/specs/2026-11-03-disc-line-design.md`. Novel canon and candidate passages: `docs/superpowers/specs/2026-11-03-disc-line-novel-canon-and-passages.md`.

```go
package characters

// RecordItemRead notes that this character has read a readable item. Reading
// it again changes nothing, and the record outlives the item.
func (c *Character) RecordItemRead(itemId int) {
	if c.ItemsRead == nil {
		c.ItemsRead = make(map[int]bool)
	}
	c.ItemsRead[itemId] = true
}

// HasReadItem reports whether this character has read the item.
func (c *Character) HasReadItem(itemId int) bool {
	return c.ItemsRead[itemId]
}

// CountItemsRead reports how many of the given items this character has read.
func (c *Character) CountItemsRead(itemIds []int) int {
	n := 0
	for _, id := range itemIds {
		if c.ItemsRead[id] {
			n++
		}
	}
	return n
}
```

```go
	const cottage, cairn, surveyCamp, lastCamp = 4023, 6349, 6351, 6365
	const survivor, alpha, sentinel = 9535, 9547, 9552

	q := loadShippedQuest(t, "76-the_disc.yaml", 76)
	require.Equal(t,
		[]string{"start", "journal", "shelter", "cairn", "camp", "alpha", "remains", "attune", "sentinel", "end"},

	keeper := loadShippedDialogue(t, "eastern_highlands", "9553.yaml")

	require.Equal(t, woken, in.CarriesItem)

	search := func(room int) EventDetails { return EventDetails{Command: "search", RoomId: room} }
	glow := func(room int) EventDetails {
		return EventDetails{Command: "cast", Noun: "chrysalis-glow", RoomId: room, TargetKind: "self"}
	}

	t.Run("walk in order", func(t *testing.T) {
		w := newQuestWalker(q).at("start")
		w.notify("command", search(lastCamp))
		require.Equal(t, "start", w.p.current, "the camp waits for the trail")

		require.Equal(t, "shelter", w.p.current)
```

```go
package guards

import (

// Every torn page counts toward The Torn Record, and every page can be had: it
// has exactly one home, on a seller's list or in a found-room trigger. A page
// added to one and left out of the other would make the achievement impossible.
func TestEveryTornPageIsCountedAndHasOneHome(t *testing.T) {
	const world = `_datafiles/world/dogmud/`

	raw, err := os.ReadFile(world + `achievements/the-torn-record.yaml`)
	if err != nil {
		t.Fatal(err)
	}
	var ach struct {
		Trigger struct {
			Type  string `yaml:"type"`
			Items []int  `yaml:"items"`
		} `yaml:"trigger"`
	}
	if err := yaml.Unmarshal(raw, &ach); err != nil {
		t.Fatal(err)
	}
	if ach.Trigger.Type != `items_read` || len(ach.Trigger.Items) < 25 {
		t.Fatalf(`the-torn-record: type %q with %d items`, ach.Trigger.Type, len(ach.Trigger.Items))
	}

	files, err := filepath.Glob(world + `items/materials-40000/*-torn_page*.yaml`)
	if err != nil {
		t.Fatal(err)
	}
	onDisk := map[string]bool{}
	for _, f := range files {
		onDisk[strings.SplitN(filepath.Base(f), `-`, 2)[0]] = true
	}
	if len(onDisk) != len(ach.Trigger.Items) {
		t.Errorf(`%d page files, %d in the achievement`, len(onDisk), len(ach.Trigger.Items))
	}

	homes := map[int]int{}
	shop := regexp.MustCompile(`(?m)^\s*- itemid: (\d+)\s*$`)
	mobFiles := []string{
		`mobs/greenford/9509-aldith_the_bookseller.yaml`,
		`mobs/the_confluence/9444-drunn_the_bookseller.yaml`,
		`mobs/new_plymouth_crafting/9332-orin_the_bookseller.yaml`,
		`mobs/marches_spur_road/250-peddler_malk.yaml`,
	}
	for _, f := range mobFiles {
		b, err := os.ReadFile(world + f)
		if err != nil {
			t.Fatal(err)
		}
		for _, m := range shop.FindAllStringSubmatch(string(b), -1) {
			homes[atoi(t, m[1])]++
		}
	}
	q, err := os.ReadFile(world + `quests/76-the_disc.yaml`)
	if err != nil {
		t.Fatal(err)
	}
	found := regexp.MustCompile(`(?m)^\s*- give_item: (\d+)\s*$`)
	for _, m := range found.FindAllStringSubmatch(string(q), -1) {
		homes[atoi(t, m[1])]++
	}
	for _, id := range ach.Trigger.Items {
		if homes[id] != 1 {
			t.Errorf(`page %d has %d homes, want 1`, id, homes[id])
		}
		spec, err := filepath.Glob(world + `items/materials-40000/` + strconv.Itoa(id) + `-*.yaml`)
		if err != nil || len(spec) != 1 {
			t.Errorf(`page %d has no item file`, id)
			continue
		}
		text, _ := os.ReadFile(spec[0])
		for _, want := range []string{`type: readable`, `not_salable: true`} {
			if !strings.Contains(string(text), want) {
				t.Errorf(`page %d: missing %q`, id, want)
			}
		}
	}
}

func atoi(t *testing.T, s string) int {
	t.Helper()
	n, err := strconv.Atoi(s)
	if err != nil {
		t.Fatal(err)
	}
	return n
}
```
